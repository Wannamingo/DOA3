cmake_minimum_required(VERSION 3.22)
project(doa3_android C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(RUNTIME "${PORT_ROOT}/recomp-runtime")
set(GAME_DIR "" CACHE PATH "Locally generated, authenticated private game code")
set(GAME_RECIPE "${PORT_ROOT}/tools/game-recipe/usa-android.json" CACHE FILEPATH "Authenticated game recipe")
set(GAME_RECIPE_SHA256 "5d03b70999092b93e8eeeea416e9696df9fa7a8bcdd7bb7927d7114d30054624" CACHE STRING "Pinned recipe identity")

if(GAME_DIR)
    file(SHA256 "${GAME_RECIPE}" recipe_sha)
    if(NOT recipe_sha STREQUAL GAME_RECIPE_SHA256)
        message(FATAL_ERROR "Game recipe identity mismatch")
    endif()
    file(READ "${GAME_RECIPE}" recipe)
    string(JSON game_xbe_sha GET "${recipe}" xbe_sha256)
    string(JSON file_count LENGTH "${recipe}" generated_files)
    math(EXPR last "${file_count} - 1")
    set(manifest "")
    set(ebp_count 0)
    set(game_sources)
    foreach(i RANGE ${last})
        string(JSON name MEMBER "${recipe}" generated_files ${i})
        string(JSON expected GET "${recipe}" generated_files "${name}")
        file(SHA256 "${GAME_DIR}/${name}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "Generated game identity mismatch: ${name}")
        endif()
        if(NOT name STREQUAL "recomp_stubs_unresolved.c")
            file(SIZE "${GAME_DIR}/${name}" size)
            string(APPEND manifest "${name}\t${size}\t${actual}\n")
        endif()
        if(name MATCHES "\\.c$" AND NOT name STREQUAL "recomp_stubs_unresolved.c")
            file(READ "${GAME_DIR}/${name}" source)
            string(LENGTH "${source}" before)
            string(REPLACE "\n    uint32_t ebp;\n" "\n    uint32_t ebp = g_seh_ebp;\n" derived "${source}")
            string(LENGTH "${derived}" after)
            math(EXPR applied "(${after} - ${before}) / 12")
            math(EXPR ebp_count "${ebp_count} + ${applied}")
            set(previous_text "")
            if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/${name}")
                file(READ "${CMAKE_CURRENT_BINARY_DIR}/${name}" previous_text)
            endif()
            if(NOT derived STREQUAL previous_text)
                file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}" "${derived}")
            endif()
            list(APPEND game_sources "${CMAKE_CURRENT_BINARY_DIR}/${name}")
        endif()
    endforeach()
    string(JSON expected_manifest GET "${recipe}" program_manifest_sha256)
    string(JSON expected_ebp GET "${recipe}" ebp_overrides)
    string(SHA256 manifest_sha "${manifest}")
    if(NOT manifest_sha STREQUAL expected_manifest OR NOT ebp_count EQUAL expected_ebp)
        message(FATAL_ERROR "Generated manifest or EBP override mismatch")
    endif()
    message(STATUS "Authenticated game manifest; ${ebp_count} existing EBP build overrides")
    add_library(doa3_game_core STATIC ${game_sources})
    set(program_definitions)
    set(program_references)
    foreach(source IN LISTS game_sources)
        file(READ "${source}" source_text)
        string(REGEX MATCHALL "void[ \t\r\n]+sub_[0-9A-Fa-f]+[ \t\r\n]*\\([ \t\r\n]*void[ \t\r\n]*\\)" matches "${source_text}")
        foreach(match IN LISTS matches)
            string(REGEX MATCH "sub_[0-9A-Fa-f]+" name "${match}")
            list(APPEND program_definitions "${name}")
        endforeach()
        string(REGEX MATCHALL "sub_[0-9A-Fa-f]+[ \t\r\n]*[(),]" references "${source_text}")
        foreach(reference IN LISTS references)
            string(REGEX MATCH "sub_[0-9A-Fa-f]+" name "${reference}")
            list(APPEND program_references "${name}")
        endforeach()
    endforeach()
    configure_file("${GAME_DIR}/recomp_funcs.h" "${CMAKE_CURRENT_BINARY_DIR}/recomp_funcs.h" COPYONLY)
    file(READ "${RUNTIME}/program_adapters.c" adapter_text)
    file(READ "${RUNTIME}/program_forwards.c" forward_text)
    set(source_text "${adapter_text}\n${forward_text}")
    string(REGEX MATCHALL "void[ \t\r\n]+sub_[0-9A-Fa-f]+[ \t\r\n]*\\([ \t\r\n]*void[ \t\r\n]*\\)" matches "${source_text}")
    foreach(match IN LISTS matches)
        string(REGEX MATCH "sub_[0-9A-Fa-f]+" name "${match}")
        list(APPEND program_definitions "${name}")
    endforeach()
    file(GLOB hand_sources "${RUNTIME}/*.c" "${RUNTIME}/*.cpp")
    list(FILTER hand_sources EXCLUDE REGEX "_test\\.(c|cpp)$|runtime_public_fixture\\.c$")
    foreach(hand_source IN LISTS hand_sources)
        file(READ "${hand_source}" hand_text)
        string(REGEX MATCHALL "sub_[0-9A-Fa-f]+[ \t\r\n]*\\(" references "${hand_text}")
        foreach(reference IN LISTS references)
            string(REGEX MATCH "sub_[0-9A-Fa-f]+" name "${reference}")
            list(APPEND program_references "${name}")
        endforeach()
    endforeach()
    list(REMOVE_DUPLICATES program_definitions)
    list(REMOVE_DUPLICATES program_references)
    set(missing_source "${CMAKE_CURRENT_BINARY_DIR}/recomp_program_missing.c")
    set(missing_text "#include <stdint.h>\nvoid recomp_program_missing(uint32_t);\n")
    set(missing_count 0)
    foreach(name IN LISTS program_references)
        if(NOT name IN_LIST program_definitions)
            string(SUBSTRING "${name}" 4 8 address)
            string(APPEND missing_text "void ${name}(void) { recomp_program_missing(0x${address}u); }\n")
            math(EXPR missing_count "${missing_count} + 1")
        endif()
    endforeach()
    set(previous_text "")
    if(EXISTS "${missing_source}")
        file(READ "${missing_source}" previous_text)
    endif()
    if(NOT missing_text STREQUAL previous_text)
        file(WRITE "${missing_source}" "${missing_text}")
    endif()
    message(STATUS "Generated ${missing_count} fail-loud absent-body stubs")
    target_sources(doa3_game_core PRIVATE "${missing_source}")
    target_include_directories(doa3_game_core PRIVATE "${RUNTIME}" "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(doa3_game_core PRIVATE -fno-strict-aliasing -ffp-contract=off -frounding-math -Wno-parentheses-equality)
    set_target_properties(doa3_game_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
